drop function if exists public.nearby_providers_directory(double precision, double precision, double precision, uuid); create function public.nearby_providers_directory( center_lat double precision, center_lng double precision, radius_km double precision default 50, p_category_id uuid default null ) returns table ( branch_id uuid, provider_id uuid, business_name text, branch_name text, locality text, whatsapp_phone text, contact_phone text, contact_email text, website_url text, description text, matricula text, category_name text, category_kind text, latitude numeric, longitude numeric, distance_km double precision, logo_url text, avg_rating numeric, review_count bigint ) language sql stable as $$ select b.id, p.id, p.business_name, b.name, b.locality, b.whatsapp_phone, p.contact_phone, p.contact_email, p.website_url, p.description, p.matricula, c.name, c.kind, b.latitude, b.longitude, ST_Distance(b.geog, ST_SetSRID(ST_MakePoint(center_lng, center_lat), 4326)::geography) / 1000.0, p.logo_url, r.avg_rating, coalesce(r.review_count, 0) from public.provider_branches b join public.providers p on p.id = b.provider_id left join public.provider_categories c on c.id = p.category_id left join public.provider_ratings_summary r on r.provider_id = p.id where b.active = true and p.active = true and p.verification_status = 'approved' and b.geog is not null and (p_category_id is null or p.category_id = p_category_id) and ST_DWithin(b.geog, ST_SetSRID(ST_MakePoint(center_lng, center_lat), 4326)::geography, radius_km * 1000) order by 16 asc; $$;
